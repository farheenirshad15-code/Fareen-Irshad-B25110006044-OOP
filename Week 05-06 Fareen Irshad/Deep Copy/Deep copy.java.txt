class Student {
    String name;
    int[] marks;

    Student(String name, int[] marks) {
        this.name = name;
        this.marks = marks;
    }

    // Deep Copy Constructor
    Student(Student s) {
        this.name = s.name;
        this.marks = new int[s.marks.length];

        for (int i = 0; i < s.marks.length; i++) {
            this.marks[i] = s.marks[i];
        }
    }
}
